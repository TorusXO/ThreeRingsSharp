cmake_minimum_required(VERSION 3.24)
project(ThreeRingsStudio VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
qt_add_executable(ThreeRingsStudio main.cpp)
target_link_libraries(ThreeRingsStudio PRIVATE Qt6::Widgets)
if(WIN32)
  target_link_libraries(ThreeRingsStudio PRIVATE dwmapi)
endif()
target_compile_definitions(ThreeRingsStudio PRIVATE TRS_VERSION="${PROJECT_VERSION}")
if(MINGW)
  target_compile_options(ThreeRingsStudio PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS ThreeRingsStudio RUNTIME DESTINATION .)
